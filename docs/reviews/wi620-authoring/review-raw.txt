"""Atomic, turn-bound primary adjudication publication.

Contracts: IF-180

Contract IF-180: carrier_candidate(root, rev) -> candidate commit or None;
    verdict_reader exposes the shared verdict API bound to that query.
    The exact commit must carry the immutable turn and pass the completion
    verifier's full reconstruction, parent, receipt and author checks. Missing
    or invalid proof returns None; callers retain the unmapped identity and
    therefore owe review. Successful proofs are cached only in this process.
    Replay may create temporary checkouts and run their bar;
    the query imports no objects and changes no primary ref, index or working tree. This is
    a read-only identity query, never permission to reset away the publication.
"""

# Cache only successful proofs in this process; restart audits them again.
_VERIFIED_PUBLICATIONS = set()


def _git(root, *args, check=True):
    p = subprocess.run(
        ["git", "-C", str(root), *args],
        capture_output=True,
        text=True,
        encoding="utf-8",
        errors="replace",
    )
    if check and p.returncode:
        raise RuntimeError((p.stdout or "") + (p.stderr or ""))
    return p.stdout.strip()


def _project(root, turn, candidate, tier="all", harness_root=None):
    primary_turn.validate_candidate(root, turn, candidate)
    plan, refusal = primary_approval.approval_projection(root, turn, candidate)
    if refusal:
        return None, None, refusal
    refusal = _apply_approval(root, candidate, plan)
    if refusal:
        return None, None, refusal
    closed, drafts, refusal = primary_close.stage_primary_adjudication(
        root, turn["spec"], judgment_path=plan["judgment"]
    )
    if refusal:
        return None, None, refusal
    minted, refusal = intake.stage_mint(
        root, drafts, "primary adjudication {}".format(turn["wi"])
    )
    if refusal:
        return None, None, refusal
    step = next(
        (
            Path(root) / rel
            for rel in (
                "scripts/trunk_step.py",
                "project-trajectory/scripts/trunk_step.py",
            )
            if (Path(root) / rel).is_file()
        ),
        None,
    )
    if step is None:
        return None, None, "trunk_step.py is absent from the candidate"
    p = subprocess.run(
        [sys.executable, str(step), "--root", ".", "--regen"],
        cwd=root,
        capture_output=True,
        text=True,
    )
    if p.returncode:
        return None, None, "regeneration failed: " + (p.stdout + p.stderr)[-2000:]
    _git(root, "add", "-A")
    product_bar = _outside_adjudication_scope(root)
    if product_bar:
        green, _out, why = publication_verify.run_bar(
            Path(root), Path(harness_root or root), tier
        )
        if not green:
            return None, None, why
        _git(root, "add", "-A")
    plan["verification"] = {
        "mode": "product-bar" if product_bar else "scope-no-product-bar",
        "paths": _git(root, "diff", "--cached", "--name-only").splitlines(),
        "result": "pass",
    }
    return closed, plan, None


def _apply_approval(root, candidate, plan):
    statuses = plan.get("statuses") or {}
    for rel, rows in statuses.items():
        live = Path(root) / rel
        refusal = intake._rewrite_toml_statuses(live, rel, list(rows))
        if refusal:
            return refusal
    if statuses or plan.get("approves"):
        # Snapshot prose carries a date. Bind it to the reviewed candidate's
        # commit date so completion replay tomorrow derives identical bytes.
        publication_day = _git(root, "show", "-s", "--format=%cs", candidate)
        clock = baseline_snapshot._today
        try:
            baseline_snapshot._today = lambda: publication_day
            approves = plan.get("approves") or {}
            baseline_snapshot.copy_live(root, approves=approves, exact_rels=approves)
        except SystemExit as exc:
            return "snapshot copy refused: {}".format(exc)
        finally:
            baseline_snapshot._today = clock
    return None


def _outside_adjudication_scope(root):
    changed = _git(root, "diff", "--cached", "--name-only").splitlines()
    allowed = (
        list(publication_verify.ADJUDICATION_SURFACES)
        + [baseline_snapshot.SNAPSHOT_DIR + "/", "docs/id-watermark"]
        + publication_verify.generated_paths(Path(root))
    )
    return not all(
        any(
            p == a.rstrip("/") or (a.endswith("/") and p.startswith(a)) for a in allowed
        )
        for p in changed
    )


def derive_tree(root, turn, candidate="HEAD", tier="all", *, import_objects=True):
    """Independently replay the exact publication and return its tree and plan."""
    root = Path(root).resolve()
    try:
        cand = _git(root, "rev-parse", candidate)
        with tempfile.TemporaryDirectory(prefix="primary-publication-") as td:
            scratch = Path(td) / "repo"
            _git(root, "clone", "--shared", "--no-checkout", str(root), str(scratch))
            _git(scratch, "checkout", "--detach", cand)
            closed, plan, refusal = _project(scratch, turn, cand, tier, root)
            if refusal:
                return None, None, None, refusal
            tree = _git(scratch, "write-tree")
            if import_objects:
                # Import newly-written blobs/tree before the disposable object store
                # disappears. A temporary commit makes the object closure fetchable;
                # no primary ref is advanced.
                carrier = _git(
                    scratch, "commit-tree", tree, "-p", cand, "-m", "projection carrier"
                )
                _git(root, "fetch", "--no-tags", str(scratch), carrier)
        return tree, closed, plan, None
    except (KeyError, OSError, RuntimeError, ValueError) as exc:
        return None, None, None, str(exc)

    """Replay, verify and atomically commit one publication; lock is caller-held."""

        if not author:

            _subject(turn), TRAILER, turn["turn"], cand, tree

        plan["verification"], sort_keys=True

    if receipt not in body.splitlines():

def carrier_candidate(root, rev):
    """Read-only mapping of this exact verified publication, never a reset peel."""
    try:
        rev, body = _git(root, "show", "-s", "--format=%H%x00%B", rev).split("\0", 1)
        matches = re.findall(
            r"(?m)^Primary-Publication-Turn: (WI-[0-9]+-[0-9a-f]{40})$", body
        )
        if len(matches) != 1:
            return None
        path = primary_turn.RECORDS + "/" + matches[0] + "/turn.toml"
        turn = primary_turn.parse_record(_git(root, "show", rev + ":" + path), path)
        return _verified_candidate(root, turn, rev)
    except (KeyError, OSError, RuntimeError, ValueError, TypeError):
        return None


def completed(root, turn):
    return publication_commit(root, turn) is not None


def pending(root):
    """One unfinished committed selection, or None; malformed/duplicate refuses."""
    root = Path(root)
    found = []
    for path in sorted((root / primary_turn.RECORDS).glob("*/turn.toml")):
        turn = primary_turn._read_record(root, path)
        if not completed(root, turn):
            found.append(turn)
    if len(found) > 1:
        raise ValueError(
            "multiple unfinished primary adjudications: "
            + ", ".join(turn["wi"] for turn in found)
        )
    return found[0] if found else None


def writer_refusal(root):
    """A durable pending turn owns the primary writer even between processes."""
    try:
        turn = pending(root)
    except (OSError, ValueError, KeyError, TypeError) as exc:
        return "primary adjudication selection is unreadable: {}".format(exc)
    if turn:
        return "primary adjudication {} is pending; resume it before another writer".format(
            turn["wi"]
        )
    return None

def git(root, *args):
    return subprocess.check_output(["git", "-C", str(root), *args], text=True).strip()


def repo(tmp_path):
    root = tmp_path / "r"
    root.mkdir()
    git(root, "init", "-q")
    git(root, "config", "user.name", "Test")
    git(root, "config", "user.email", "test@example.invalid")
    (root / "docs/work/queued").mkdir(parents=True)
    (root / "docs/work/queued/WI-1-x.md").write_text("x\n")
    git(root, "add", "-A")
    git(root, "commit", "-qm", "base")

def test_project_runs_candidate_bar_with_primary_harness_python(tmp_path, monkeypatch):
    """Product effects run candidate check.py through the primary checkout venv."""
    primary = tmp_path / "primary"
    scratch = tmp_path / "scratch"
    primary.mkdir()
    scratch.mkdir()
    for root in (primary, scratch):
        git(root, "init", "-q")
        git(root, "config", "user.name", "Test")
        git(root, "config", "user.email", "test@example.invalid")
    sentinel = primary / "harness-used"
    py = primary / ".venv/bin/python"
    py.parent.mkdir(parents=True)
    py.write_text(
        '#!/bin/sh\nprintf used > {!r}\nexec {!r} "$@"\n'.format(
            str(sentinel), sys.executable
        ),
        encoding="utf-8",
    )
    py.chmod(0o755)
    scripts = scratch / "scripts"
    scripts.mkdir()
    (scripts / "trunk_step.py").write_text("pass\n", encoding="utf-8")
    (scripts / "check.py").write_text(
        "import pathlib\n"
        "assert pathlib.Path('candidate-only.txt').is_file()\n"
        "print('  PASS  candidate-check   0.0s')\n",
        encoding="utf-8",
    )
    (scratch / "candidate-only.txt").write_text("product effect\n", encoding="utf-8")
    git(scratch, "add", "-A")
    git(scratch, "commit", "-qm", "candidate")
    candidate = git(scratch, "rev-parse", "HEAD")

    monkeypatch.setattr(pp.primary_turn, "validate_candidate", lambda *_a: None)
    monkeypatch.setattr(
        pp.primary_approval,
        "approval_projection",
        lambda *_a, **_k: (
            {
                "outcome": "return",
                "statuses": {},
                "approves": {},
                "author": None,
                "findings": [],
                "judgment": "docs/reviews/judgment.md",
            },
            None,
        ),
    )
    monkeypatch.setattr(pp, "_apply_approval", lambda *_a: None)

    def stage_product(root, *_a, **_k):
        (Path(root) / "candidate-only.txt").write_text(
            "published product effect\n", encoding="utf-8"
        )
        git(root, "add", "candidate-only.txt")
        return ["WI-1"], [], None

    monkeypatch.setattr(pp.primary_close, "stage_primary_adjudication", stage_product)
    monkeypatch.setattr(pp.intake, "stage_mint", lambda *_a, **_k: ([], None))

    closed, plan, refusal = pp._project(
        scratch,
        {"wi": "WI-1", "spec": "docs/work/queued/WI-1.md"},
        candidate,
        "smoke",
        primary,
    )

    assert refusal is None and closed == ["WI-1"]
    assert plan["verification"]["mode"] == "product-bar"
    assert sentinel.read_text(encoding="utf-8") == "used"


@pytest.mark.parametrize("product_bar", [False, True], ids=["scoped", "required-bar"])
def test_real_return_publish_and_completion_replay(tmp_path, monkeypatch, product_bar):
    root = repo(tmp_path)
    spec = root / "docs/work/queued/WI-1-x.md"
    spec.write_text(
        '+++\nid = "WI-1"\ntitle = "clean close"\nsafety_class = "adjudication"\nspecref = "x"\n+++\n\n## Context\n\njudge\n'
    )
    step = root / "project-trajectory/scripts/trunk_step.py"
    step.parent.mkdir(parents=True)
    step.write_text("import sys\nsys.exit(0)\n")
    if product_bar:
        step.write_text(
            "from pathlib import Path\n"
            "Path('derived-product').write_text('regenerated\\n')\n"
        )
        (step.parent / "check.py").write_text(
            "from pathlib import Path\n"
            "assert Path('derived-product').read_text() == 'regenerated\\n'\n"
            "Path('bar-ran').write_text('checked\\n')\n"
            "print('  PASS  candidate-check   0.0s')\n"
        )
    git(root, "add", "-A")
    git(root, "commit", "-qm", "base")
    turn = primary_adjudication.begin(root, "WI-1")
    base = turn["base"]
    turn_id = turn["turn"]
    proposal = verdict.tree_identity(root, base)
    approval = (
        '\n## Approval\n\n```toml\nversion = 1\nturn = "{}"\n'
        'base = "{}"\nproposal = "{}"\ndecision = []\n```\n'
    ).format(turn_id, base, proposal)
    spec.write_text(spec.read_text() + approval)
    log = root / "docs/iteration/WI-1.log"
    log.parent.mkdir(parents=True)
    log.write_text(
        "# phase: ADJUDICATE\n# wi: WI-1\n# base: {}\n# provider: test\n# roster-row: test\n# model: test\n# session-id: s\n# invocation-id: i\n".format(
            base[:12]
        )
    )
    review = root / "docs/reviews/WI-1-001-REVIEW-A.md"
    review.parent.mkdir(parents=True, exist_ok=True)
    review.write_text("VERDICT: APPROVE\nOUTCOME: REVIEWED\n")
    git(root, "add", "-A")
    git(root, "commit", "-qm", "ADJUDICATE verdict")
    candidate = git(root, "rev-parse", "HEAD")
    log.write_text(
        log.read_text()
        + "# primary-turn: {}\n# judgment: docs/reviews/WI-1-001-REVIEW-A.md\n# candidate: {}\n".format(
            turn_id, candidate
        )
    )
    git(root, "add", "-A")
    git(root, "commit", "-qm", "ADJUDICATE telemetry")
    candidate = git(root, "rev-parse", "HEAD")
    hook = root / ".git/hooks/pre-commit"
    hook.write_text("#!/bin/sh\nexit 1\n")
    hook.chmod(0o755)
    failed, refusal = pp.publish(root, turn, candidate)
    assert failed is None and "commit failed" in refusal
    assert git(root, "rev-parse", "HEAD") == candidate
    assert git(root, "status", "--porcelain") == ""
    hook.unlink()
    hook.write_text(
        "#!/bin/sh\nprintf 'smuggled\\n' > hook-smuggled\ngit add hook-smuggled\n",
        encoding="utf-8",
    )
    hook.chmod(0o755)
    closed, refusal = pp.publish(root, turn, candidate)
    assert closed is None
    assert refusal == "publication commit tree differs from the verified tree"
    assert git(root, "rev-parse", "HEAD") == candidate
    assert git(root, "status", "--porcelain") == ""
    assert not (root / "hook-smuggled").exists()
    assert not pp.completed(root, turn)
    hook.unlink()
    closed, refusal = pp.publish(root, turn, candidate)
    assert refusal is None and closed == ["WI-1"]
    assert (root / "docs/archive/work/complete/WI-1-x.md").is_file()
    landed = git(root, "rev-parse", "HEAD")
    assert pp.completed(root, turn) and pp.publication_commit(root, turn) == landed
    assert pp.completed(root, turn)  # exact replay is idempotent; no new commit
    assert git(root, "rev-parse", "HEAD") == landed
    # Both readers use the same verified carrier; the reset reader must not.
    identity = verdict.governing_identity(
        root, turn["primary_branch"], candidate, carrier=pp.carrier_candidate
    )
    # A cold query must preserve dirty as well as clean primary state. Compare
    # every file, including index, refs, worktree, objects and FETCH_HEAD.
    if product_bar:
        assert (root / "bar-ran").read_text() == "checked\n"
    (root / "staged-sentinel").write_text("staged\n")
    git(root, "add", "staged-sentinel")
    (root / "staged-sentinel").write_text("unstaged over staged\n")
    (root / "untracked-sentinel").write_text("untracked\n")
    before = {
        p.relative_to(root): p.read_bytes() for p in root.rglob("*") if p.is_file()
    }
    pp._VERIFIED_PUBLICATIONS.clear()
    assert pp.carrier_candidate(root, landed) == candidate
    after = {
        p.relative_to(root): p.read_bytes() for p in root.rglob("*") if p.is_file()
    }
    assert after == before
    git(root, "reset", "--hard", landed)
    (root / "untracked-sentinel").unlink()
    assert (
        verdict.governing_identity(
            root, turn["primary_branch"], carrier=pp.carrier_candidate
        )
        == identity
    )
    assert verdict.work_tip(root, turn["primary_branch"]) == landed
    message = git(root, "show", "-s", "--format=%B", landed)
    final_tree = git(root, "rev-parse", landed + "^{tree}")
    candidate_tree = git(root, "rev-parse", candidate + "^{tree}")
    for forged_message, forged_tree, parent in (
        (message, candidate_tree, candidate),  # missing the derived effects
        (message, final_tree, landed),  # copied true words onto later work
        (message.replace(turn_id, "WI-999-" + base), final_tree, candidate),
        (message.replace(candidate, base), final_tree, candidate),
        (
            message.replace('"result": "pass"', '"result": "fail"'),
            final_tree,
            candidate,
        ),
        (message + "\n" + pp.TRAILER + turn_id, final_tree, candidate),
    ):
        forged = git(
            root, "commit-tree", forged_tree, "-p", parent, "-m", forged_message
        )
        assert pp.carrier_candidate(root, forged) is None
        if forged_message == message + "\n" + pp.TRAILER + turn_id:
            assert pp._verified_candidate(root, turn, forged) is None
    bad_author = subprocess.run(
        [
            "git",
            "-C",
            str(root),
            "commit-tree",
            final_tree,
            "-p",
            candidate,
            "-m",
            message,
        ],
        env=dict(
            os.environ,
            GIT_AUTHOR_NAME="Wrong author",
            GIT_AUTHOR_EMAIL="wrong@example.invalid",
        ),
        capture_output=True,
        text=True,
        check=True,
    ).stdout.strip()
    assert pp.carrier_candidate(root, bad_author) is None
    original_derive = pp.derive_tree
    calls = []

    def counted_derive(*args, **kwargs):
        calls.append(args[2])
        return original_derive(*args, **kwargs)

    monkeypatch.setattr(pp, "derive_tree", counted_derive)
    assert pp.completed(root, turn)
    assert pp.pending(root) is None
    assert calls == []
    mismatched_turn = dict(turn, primary_root="wrong-primary-root")
    assert not pp.completed(root, mismatched_turn)
    assert calls == [candidate]
    calls.clear()
    forged_tree = git(root, "show", "-s", "--format=%T", "HEAD")
    git(
        root,
        "commit",
        "--allow-empty",
        "-qm",
        "adjudicate: WI-1 primary publication\n\n{}{}\nPublication-Candidate: {}\n"
        "Publication-Tree: {}".format(pp.TRAILER, turn["turn"], landed, forged_tree),
    )
    assert pp.completed(root, turn)
    assert calls == [landed]


def test_bootstrapped_return_publication_replay_keeps_real_regeneration_fresh(
    tmp_path,
):
    root = tmp_path / "scaffold"
    root.mkdir()
    boot = run_py([SCRIPTS / "bootstrap.py", "--dest", root], cwd=tmp_path)
    assert boot.returncode == 0, boot.stdout + boot.stderr
    make_minimal_project(root)
    set_process_key(root, "policies", "review_rounds", 0)
    spec = root / "docs/work/queued/WI-1-return.md"
    spec.parent.mkdir(parents=True, exist_ok=True)
    spec.write_text(
        '+++\nid = "WI-1"\ntitle = "record a finite return"\n'
        'safety_class = "adjudication"\nspecref = "docs/log.md"\n+++\n'
        "\n## Context\n\nJudge the empty scope.\n",
        encoding="utf-8",
    )
    record_ids(root)
    git(root, "init", "-q")
    git(root, "config", "user.name", "Test")
    git(root, "config", "user.email", "test@example.invalid")
    git(root, "add", "-A")
    git(root, "commit", "-qm", "seed bootstrapped primary")
    turn = primary_adjudication.begin(root, "WI-1")
    base = turn["base"]
    verdict_path = root / "docs/reviews/WI-1-primary.md"
    verdict_path.parent.mkdir(parents=True, exist_ok=True)
    verdict_path.write_text("OUTCOME: RETURN\n", encoding="utf-8")
    git(root, "add", "-A")
    git(root, "commit", "-qm", "typed primary return")
    spec.write_text(
        spec.read_text(encoding="utf-8")
        + '\n## Approval\n\n```toml\nversion = 1\nturn = "{}"\n'
        'base = "{}"\nproposal = "{}"\ndecision = []\n```\n'.format(
            turn["turn"], base, verdict.tree_identity(root, base)
        ),
        encoding="utf-8",
    )
    git(root, "add", "-A")
    git(root, "commit", "-qm", "record primary return payload")
    reviewed = git(root, "rev-parse", "HEAD")
    telemetry = root / "docs/iteration/WI-1-primary.log"
    telemetry.parent.mkdir(parents=True, exist_ok=True)
    telemetry.write_text(
        "# phase: ADJUDICATE\n# wi: WI-1\n# provider: test\n"
        "# roster-row: route\n# model: model\n# session-id: session\n"
        "# invocation-id: invocation\n# base: {}\n# primary-turn: {}\n"
        "# judgment: docs/reviews/WI-1-primary.md\n# candidate: {}\n# ---\n".format(
            base[:12], turn["turn"], reviewed
        ),
        encoding="utf-8",
    )
    git(root, "add", "-A")
    git(root, "commit", "-qm", "bind primary return telemetry")
    candidate = git(root, "rev-parse", "HEAD")
    linked = tmp_path / "linked-worker"
    git(root, "worktree", "add", "--detach", str(linked), candidate)
    primary_head = git(root, "rev-parse", "HEAD")
    linked_head = git(linked, "rev-parse", "HEAD")

    closed, refusal = pp.publish(linked, turn, candidate)

    assert closed is None
    assert refusal == "adjudication needs the designated primary checkout"
    assert git(root, "rev-parse", "HEAD") == primary_head
    assert git(linked, "rev-parse", "HEAD") == linked_head
    assert git(root, "status", "--porcelain") == ""
    assert git(linked, "status", "--porcelain") == ""

    closed, refusal = pp.publish(root, turn, candidate)

    assert refusal is None and closed == ["WI-1"]
    landed = git(root, "rev-parse", "HEAD")
    assert pp.completed(root, turn)
    assert git(root, "rev-parse", "HEAD") == landed
    for script in ("derive_stage.py", "gen_trajectory.py"):
        check = subprocess.run(
            [sys.executable, "scripts/" + script, "--check"],
            cwd=root,
            capture_output=True,
            text=True,
        )
        assert check.returncode == 0, check.stdout + check.stderr
    assert git(root, "status", "--porcelain") == ""
    before = {
        path: path.read_bytes()
        for path in root.rglob("*")
        if path.is_file() and ".git" not in path.parts
    }
    regen = subprocess.run(
        [sys.executable, "scripts/trunk_step.py", "--root", ".", "--regen"],
        cwd=root,
        capture_output=True,
        text=True,
    )
    assert regen.returncode == 0, regen.stdout + regen.stderr
    after = {path: path.read_bytes() for path in before}
    dashboard = root / "PROJECT_STATE.html"
    stage = root / "docs/stage"
    # Both writers declare their revision/date comments informational: their
    # --check paths compare dashboard content without ASOF_RE and stage through
    # field_block, because a commit necessarily changes its own as-of identity.
    assert {path for path in before if before[path] != after[path]} == {
        dashboard,
        stage,
    }
    assert gen_trajectory.ASOF_RE.sub("", after[dashboard].decode()) == (
        gen_trajectory.ASOF_RE.sub("", before[dashboard].decode())
    )
    assert kitstage.field_block(kitstage.parse(after[stage].decode())) == (
        kitstage.field_block(kitstage.parse(before[stage].decode()))
    )
    assert pp.completed(root, turn)
    git(root, "checkout", "--", "PROJECT_STATE.html", "docs/stage")
    assert git(root, "status", "--porcelain") == ""

    )

    evidence_file.write_text("OUTCOME: APPROVE rows=1\n")

    )

    git(root, "commit", "-qm", "ADJUDICATE telemetry")
    candidate = git(root, "rev-parse", "HEAD")
    return root, turn, candidate, registry, snapshot

    assert pp.completed(root, turn)  # replay is independent of today's date

    # receipt and adjudicator author, so another invalid field cannot mask it.

def _multiline_delims(text):
    """How many multi-line-string delimiters this line opens or closes, per
    delimiter kind — the minimum TOML awareness a line rewrite needs.

    Counted on the whole line, comment text included: a delimiter inside a `#`
    comment is not a string opener, but a `#` inside a string IS just text, and
    of the two ways to be wrong only treating a delimiter as significant is
    safe. An ODD count toggles the state; an even one — opened and closed on the
    same line, which is how a single-line triple-quoted cell reads — does not.
    """
    return {d: text.count(d) for d in _TOML_MULTILINE}


def _flip_status_lines(lines, table, rid):
    """Rewrite `[<table>.<rid>]`'s `status = ...` line to `Approved`, in place.
    True when a line moved. A LINE REWRITE ON `bootstrap.set_process_key`'s
    PATTERN, and for its reasons: stdlib has no TOML
    writer, and re-serialising the registry to change one cell would normalise
    away every comment and the file's authored ordering — a whole-file diff for
    a one-word act, on the registry whose diffs the amendment guard reads.

    IT TRACKS MULTI-LINE STRING STATE, and that is not defensive tidiness — it
    is the difference between an approval and a corruption. The spine's prose
    cells are `\"\"\"...\"\"\"` blocks that quote registry syntax freely, so a
    requirement can contain a line whose text reads `status = ...`. Rewriting by
    physical line alone edited THAT line, left the row's real `status` at
    `Modified`, and returned True — so the tool reported a flip it had not made
    while silently rewriting attested requirement text. A line inside a string
    is DATA; only a line at the table's top level is a key.

    The same state tracking is what makes the table-header scan sound: a `[` in
    the middle of a prose cell would otherwise read as the start of a new row
    and end the search early."""
    header = "[{}.{}]".format(table, rid)
    in_row = False
    open_delim = None  # the multi-line delimiter we are currently inside, if any
    for i, line in enumerate(lines):
        counts = _multiline_delims(line)
        if open_delim is not None:
            # Inside a multi-line string: this line is DATA. It can only end the
            # string, never open a table or set a key.
            if counts[open_delim] % 2:
                open_delim = None
            continue
        stripped = line.strip()
        opened = next((d for d in _TOML_MULTILINE if counts[d] % 2), None)
        if stripped.startswith("[") and stripped.endswith("]") and opened is None:
            in_row = stripped == header
            continue
        if in_row and stripped.split("=")[0].strip() == _STATUS_KEY and opened is None:
            lines[i] = '{} = "Approved"'.format(_STATUS_KEY)
            return True
        open_delim = opened
    return False


def _rewrite_toml_statuses(live, rel, ids):
    """Move named status lines to Approved; return a refusal string or None.

    Split out of `_apply_flips` when the snapshot copy joined it (D-9 step 3):
    the carrier-specific write is a self-contained job with its own refusal, and
    keeping it inline made the caller's branch count grow every time a step was
    added to an act that is really "flip, then record what was flipped".

    The file's OWN newline style is preserved. The contract this writer
    advertises is that every byte except the one status cell is unchanged, and
    silently converting a CRLF registry to LF makes a one-word approval a
    whole-file diff — on exactly the registry whose diffs the amendment guard
    reads. `newline=""` keeps the bytes; the split is on the detected
    terminator."""
    # open() rather than Path.read_text: read_text only grew a `newline`
    # parameter at Python 3.13, and the kit's floor is 3.11 — on the floor the
    # keyword is a TypeError, found 2026-08-15 when the suite first ran on the
    # repo's own 3.11.9 venv (the sitting-sweep log entry carries the account).
    with open(live, "r", encoding="utf-8-sig", newline="") as fh:
        raw = fh.read()
    eol = "\r\n" if "\r\n" in raw else "\n"
    lines = raw.split(eol)

            return (

                "report a flip that was not written".format(

                )
            )
    live.write_text(eol.join(lines), encoding="utf-8", newline="")
